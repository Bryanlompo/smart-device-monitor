cmake_minimum_required(VERSION 3.16)
project(smart_device_monitor VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(monitor_common INTERFACE)
target_include_directories(monitor_common INTERFACE common/include)
add_executable(device_simulator simulator/main.cpp)
target_link_libraries(device_simulator PRIVATE monitor_common)
include(CTest)
if(BUILD_TESTING)
  add_executable(monitor_tests tests/test_monitor.cpp)
  target_link_libraries(monitor_tests PRIVATE monitor_common)
  add_test(NAME monitor_tests COMMAND monitor_tests)
endif()
